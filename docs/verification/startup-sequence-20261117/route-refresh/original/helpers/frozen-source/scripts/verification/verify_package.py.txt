def sha256(path):
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def require(condition, message):
    if not condition:
        raise ValueError(message)


def record(report, name, **details):
    report["checks"].append({"name": name, "status": "passed", **details})


def check_spatial_dither_notices(archive, app_root, report):
    """Retain frozen license/provenance bytes without distributing research assets."""
    expected = {SPATIAL_DITHER_PREFIX + name for name in SPATIAL_DITHER_NOTICES}
    actual = {name for name in archive.namelist() if name.startswith(SPATIAL_DITHER_PREFIX)}
    require(actual == expected, "The spatial-dither notice group is incomplete or contains research inputs")
    reference = app_root / SPATIAL_DITHER_REFERENCE
    contents = {}
    for name in SPATIAL_DITHER_NOTICES:
        content = archive.read(SPATIAL_DITHER_PREFIX + name)
        require(content == (reference / name).read_bytes(),
                f"A retained spatial-dither notice differs from the frozen source: {name}")
        contents[name] = content
    upstream = json.loads(contents["UPSTREAM_FILES.json"])
    licenses = [item for item in upstream.get("files", []) if item.get("path") == "LICENSE.AIDOKU"]
    require(len(licenses) == 1 and licenses[0].get("bytes") == len(contents["LICENSE.AIDOKU"])
            and licenses[0].get("sha256") == hashlib.sha256(contents["LICENSE.AIDOKU"]).hexdigest(),
            "The spatial-dither license differs from its retained source manifest")
    record(report, "spatial_dither_notice_provenance", retained_files=list(SPATIAL_DITHER_NOTICES),
           source_repository=upstream.get("source_repository"),
           source_snapshot_sha256=upstream.get("source_snapshot_sha256"),
           reference_images_bundled=False,
           scope="Original source/license provenance only; reference-project verification is separate")


def check_startup_notices(archive, app_root, report):
    """Bind retained startup notices to the pinned, selected upstream sources."""
    expected = {STARTUP_PREFIX + name for name in STARTUP_NOTICES}
    actual = {name for name in archive.namelist() if name.startswith(STARTUP_PREFIX)}
    require(actual == expected, "The startup notice group is incomplete or contains source inputs")
    reference = app_root / STARTUP_REFERENCE
    contents = {}
    for name in STARTUP_NOTICES:
        content = archive.read(STARTUP_PREFIX + name)
        require(content == (reference / name).read_bytes(),
                f"A retained startup notice differs from the frozen source: {name}")
        contents[name] = content
    upstream = json.loads(contents["UPSTREAM_FILES.json"])
    require(upstream.get("source_repository") == "AidokuForKindle"
            and upstream.get("source_commits") == list(STARTUP_COMMITS)
            and upstream.get("license") == "GPL-3.0-or-later",
            "The startup provenance differs from the pinned source identities")
    files = upstream.get("files")
    require(isinstance(files, list) and len(files) == len(STARTUP_SOURCES)
            and all(isinstance(item, dict) and isinstance(item.get("path"), str) for item in files),
            "The startup source manifest is incomplete or malformed")
    paths = [item["path"] for item in files]
    require(len(set(paths)) == len(paths) and set(paths) == set(STARTUP_SOURCES),
            "The startup source manifest has duplicate, missing, or unexpected paths")
    for item in files:
        relative = item["path"]
        commit, source_path = STARTUP_SOURCES[relative]
        require(item.get("source_commit") == commit and item.get("source_path") == source_path,
                f"A startup source identity differs from its pinned input: {relative}")
        path = reference / relative
        require(path.is_file() and not path.is_symlink(), f"A retained startup source is unavailable: {relative}")
        path.resolve().relative_to(reference.resolve())
        data = path.read_bytes()
        require(item.get("bytes") == len(data) and item.get("sha256") == hashlib.sha256(data).hexdigest(),
                f"A retained startup source differs from its manifest: {relative}")
    require(hashlib.sha256(contents["LICENSE.AIDOKU"]).hexdigest() == STARTUP_LICENSE_SHA256,
            "The startup license differs from the original GPLv3 text")
    record(report, "startup_implementation_notice_provenance", retained_files=list(STARTUP_NOTICES),
           source_repository=upstream["source_repository"], source_commits=list(STARTUP_COMMITS),
           source_snapshots=len(STARTUP_SOURCES) - 1, source_code_bundled=False,
           scope="Selected original sources and license only; independent BiliComics verification required")


def check_startup_sequence_notices(archive, app_root, report):
    """Keep the item 1 candidate reference separate from earlier startup records."""
    expected = {STARTUP_SEQUENCE_PREFIX + name for name in STARTUP_SEQUENCE_NOTICES}
    actual = {name for name in archive.namelist() if name.startswith(STARTUP_SEQUENCE_PREFIX)}
    require(actual == expected, "The startup-sequence notice group is incomplete or contains source inputs")
    reference = app_root / STARTUP_SEQUENCE_REFERENCE
    contents = {}
    for name in STARTUP_SEQUENCE_NOTICES:
        content = archive.read(STARTUP_SEQUENCE_PREFIX + name)
        require(content == (reference / name).read_bytes(),
                f"A retained startup-sequence notice differs from the frozen source: {name}")
        contents[name] = content
    upstream = json.loads(contents["UPSTREAM_FILES.json"])
    require(isinstance(upstream, dict) and upstream.get("schema_version") == 1
            and upstream.get("source_repository") == "AidokuForKindle"
            and upstream.get("source_repository_url") == "https://github.com/moooyo/aidokuForKindle"
            and upstream.get("source_commits") == [STARTUP_SEQUENCE_COMMIT]
            and upstream.get("production_source_commit") == STARTUP_SEQUENCE_PRODUCTION_COMMIT
            and upstream.get("license") == "GPL-3.0-or-later",
            "The startup-sequence provenance differs from the pinned source identities")
    require(hashlib.sha256(contents["LICENSE.AIDOKU"]).hexdigest()
            == STARTUP_SEQUENCE_SOURCES["LICENSE.AIDOKU"][2],
            "The startup-sequence license differs from the original GPLv3 text")
    files = upstream.get("files")
    require(isinstance(files, list) and len(files) == len(STARTUP_SEQUENCE_SOURCES)
            and all(isinstance(item, dict) and isinstance(item.get("path"), str) for item in files),
            "The startup-sequence source manifest is incomplete or malformed")
    paths = [item["path"] for item in files]
    require(len(set(paths)) == len(paths) and set(paths) == set(STARTUP_SEQUENCE_SOURCES),
            "The startup-sequence source manifest has duplicate, missing, or unexpected paths")
    for item in files:
        relative = item["path"]
        source_path, blob, digest = STARTUP_SEQUENCE_SOURCES[relative]
        require(item.get("source_commit") == STARTUP_SEQUENCE_COMMIT
                and item.get("source_path") == source_path
                and item.get("source_blob") == blob and item.get("production_source_blob") == blob,
                f"A startup-sequence source identity differs from its pinned input: {relative}")
        path = reference / relative
        require(path.is_file() and not path.is_symlink(),
                f"A retained startup-sequence source is unavailable: {relative}")
        path.resolve().relative_to(reference.resolve())
        data = path.read_bytes()
        actual_digest = hashlib.sha256(data).hexdigest()
        require(item.get("bytes") == len(data) and item.get("sha256") == actual_digest,
                f"A retained startup-sequence source differs from its manifest: {relative}")
        require(actual_digest == digest,
                f"A retained startup-sequence source differs from its pinned bytes: {relative}")
    record(report, "startup_sequence_notice_provenance", retained_files=list(STARTUP_SEQUENCE_NOTICES),
           source_repository=upstream["source_repository"], source_commits=[STARTUP_SEQUENCE_COMMIT],
           production_source_commit=STARTUP_SEQUENCE_PRODUCTION_COMMIT,
           source_snapshots=len(STARTUP_SEQUENCE_SOURCES) - 1, source_code_bundled=False,
           scope="Item 1 source/license reference only; no final adoption, performance or device claim")


def check_package(args, report):
    archive_path = args.archive.resolve(strict=True)
    binary = args.binary.resolve(strict=True)
    worker = (args.worker or binary.with_name("kindle-jpeg-worker")).resolve(strict=True)
    app_root = args.app_root.resolve(strict=True)
    manifest_path = archive_path.with_suffix(".manifest.json")
    require(app_root.is_dir(), "The application source root is not a directory")
    require(archive_path.is_file() and binary.is_file() and worker.is_file(),
            "The archive, application, and JPEG worker must be regular files")
    require(manifest_path.is_file() and not manifest_path.is_symlink(), "The archive manifest is unavailable or is a symlink")
    binary_digest = sha256(binary)
    worker_digest = sha256(worker)
    archive_digest = sha256(archive_path)
    require(binary_digest == args.expected_binary_sha256.lower(), "The ARM binary SHA-256 differs from the selected input")
    if args.expected_worker_sha256:
        require(worker_digest == args.expected_worker_sha256.lower(), "The JPEG worker SHA-256 differs from the selected input")
    if args.expected_archive_sha256:
        require(archive_digest == args.expected_archive_sha256.lower(), "The archive SHA-256 differs from the selected input")
    report["inputs"] = {
        "archive": str(archive_path),
        "archive_bytes": archive_path.stat().st_size,
        "archive_sha256": archive_digest,
        "manifest": str(manifest_path),
        "manifest_sha256": sha256(manifest_path),
        "binary": str(binary),
        "binary_bytes": binary.stat().st_size,
        "binary_sha256": binary_digest,
        "worker": str(worker),
        "worker_bytes": worker.stat().st_size,
        "worker_sha256": worker_digest,
        "app_root": str(app_root),
    }
    record(report, "selected_input_digests", archive_digest_explicitly_selected=bool(args.expected_archive_sha256),
           worker_digest_explicitly_selected=bool(args.expected_worker_sha256))

    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    require(manifest.get("development_candidate") is True and manifest.get("device_verified") is False,
            "The sidecar must identify an unverified development candidate")
    files = manifest.get("files")
    require(isinstance(files, list) and len(files) == len(ALLOWED_FILES), "The manifest does not match the exact allowed entry count")
    require(all(isinstance(item, dict) and isinstance(item.get("path"), str) for item in files),
            "The manifest contains an invalid entry")
    manifest_names = [item["path"] for item in files]
    require(len(set(manifest_names)) == len(files) and set(manifest_names) == ALLOWED_FILES,
            "The manifest has duplicate, missing, or unexpected paths")

    with zipfile.ZipFile(archive_path) as archive:
        names = archive.namelist()
        require(len(names) == len(set(names)) == len(ALLOWED_FILES) and set(names) == ALLOWED_FILES,
                "The archive has duplicate, missing, or unexpected paths")
        require(archive.testzip() is None, "The archive failed its CRC check")
        record(report, "archive_crc_and_exact_allowlist", entries=len(names), runtime_account_files_present=False)
        entries = []
        for item in files:
            name = item["path"]
            info = archive.getinfo(name)
            mode = (info.external_attr >> 16) & 0xFFFF
            require(not info.is_dir() and not stat.S_ISLNK(mode), f"The archive entry is not a regular file: {name}")
            require(info.date_time == FIXED_TIMESTAMP, f"The archive entry timestamp is not fixed: {name}")
            data = archive.read(name)
            digest = hashlib.sha256(data).hexdigest()
            require(len(data) == item.get("bytes") and digest == item.get("sha256"),
                    f"The archive entry differs from its manifest: {name}")
            if name in EXECUTABLE_FILES:
                require(stat.S_IMODE(mode) == 0o755, f"The executable mode is not 0755: {name}")
            entries.append({"path": name, "bytes": len(data), "sha256": digest,
                            "mode": oct(stat.S_IMODE(mode)), "fixed_timestamp": True})
        report["archive_entries"] = entries
        record(report, "entry_sizes_digests_timestamps_and_modes", executable_modes="0755")

        embedded = archive.read("bilicomics/app/bilicomics")
        require(len(embedded) == binary.stat().st_size and hashlib.sha256(embedded).hexdigest() == binary_digest,
                "The packaged executable differs from the selected ARM binary")
        record(report, "packaged_binary_matches_selected_binary")
        embedded_worker = archive.read("bilicomics/app/kindle-jpeg-worker")
        require(len(embedded_worker) == worker.stat().st_size
                and hashlib.sha256(embedded_worker).hexdigest() == worker_digest,
                "The packaged JPEG worker differs from the selected ARM binary")
        record(report, "packaged_jpeg_worker_matches_selected_binary")
        check_spatial_dither_notices(archive, app_root, report)
        check_startup_notices(archive, app_root, report)
        check_startup_sequence_notices(archive, app_root, report)
        accepted = [line for line in archive.read("bilicomics/data/device-allowlist").decode("utf-8").splitlines()
                    if line.strip() and not line.lstrip().startswith("#")]
        require(not accepted, "The development archive unexpectedly accepts a device/firmware combination")
        record(report, "device_allowlist_empty", accepted_entries=0)

        startup_manifest = json.loads(archive.read("notices/fonts/startup-manifest.json"))
        startup_font = app_root / "assets/fonts/StartupSansSC.otf"
        output = startup_manifest.get("output", {})
        require(output.get("sha256") == sha256(startup_font) and output.get("bytes") == startup_font.stat().st_size,
                "The startup font differs from the packaged provenance")
        retained = startup_manifest.get("retained_license_files")
        require(isinstance(retained, dict) and retained, "The startup font has no retained license provenance")
        for name, identity in retained.items():
            data = archive.read("notices/fonts/" + name)
            require(isinstance(identity, dict) and len(data) == identity.get("bytes")
                    and hashlib.sha256(data).hexdigest() == identity.get("sha256"),
                    f"A retained startup-font notice differs from its provenance: {name}")
        record(report, "startup_font_and_notice_provenance", font_sha256=sha256(startup_font),
               font_bytes=startup_font.stat().st_size, retained_notices=sorted(retained))

        jpeg_notices = {
            "LICENSE-APACHE": "LICENSE-APACHE", "LICENSE-MIT": "LICENSE-MIT", "LICENSE-ZLIB": "LICENSE-ZLIB",
            "UPSTREAM.md": "UPSTREAM.md", "PATCHES.md": "PATCHES.md",
            "0001-strict-entropy-completeness.patch": "patches/0001-strict-entropy-completeness.patch",
        }
        for packaged, source in jpeg_notices.items():
            require(archive.read("notices/jpeg/" + packaged) == (app_root / "vendor/zune-jpeg" / source).read_bytes(),
                    f"A retained JPEG decoder notice differs from the selected source: {packaged}")
        record(report, "jpeg_decoder_notice_provenance", retained_files=sorted(jpeg_notices))

        triangle_notices = ("LICENSE-MIT", "LICENSE-APACHE", "README.md", "SHA256SUMS")
        for name in triangle_notices:
            require(archive.read("notices/image-triangle/" + name)
                    == (app_root / "vendor/image-triangle-reference-0.25.10" / name).read_bytes(),
                    f"A retained Triangle resize notice differs from the selected source: {name}")
        record(report, "triangle_resize_notice_provenance", retained_files=list(triangle_notices))

        media_notices = ("LICENSE-GPL-3.0-or-later", "README.md", "UPSTREAM_FILES.json")
        for name in media_notices:
            require(archive.read("notices/aidoku-native-media/" + name)
                    == (app_root / "vendor/aidoku-native-media" / name).read_bytes(),
                    f"A retained native-media notice differs from the selected source: {name}")
        record(report, "native_media_notice_provenance", retained_files=list(media_notices))

    for role, executable, suffix in (("", binary, ".readelf.txt"), ("jpeg_worker_", worker, ".worker.readelf.txt")):
        with executable.open("rb") as stream:
            header = stream.read(64)
        require(len(header) >= 52 and header[:7] == b"\x7fELF\x01\x01\x01",
                f"The {role}executable is not ELF32 little-endian version 1")
        machine = struct.unpack_from("<H", header, 18)[0]
        flags = struct.unpack_from("<I", header, 36)[0]
        require(machine == 40, f"The {role}executable is not an ARM ELF")
        require(flags >> 24 == 5, f"The {role}executable does not declare ARM EABI5")
        require(flags & 0x400 and not flags & 0x200, f"The {role}executable does not declare the ARM hard-float ABI")
        record(report, role + "arm_elf_header", elf_class=32, byte_order="little", machine="ARM", eabi_version=5, hard_float=True)

        readelf_parts = []
        environment = os.environ.copy()
        environment["LC_ALL"] = "C"
        for option in ("-h", "-l", "-d"):
            command = ["readelf", option, str(executable)]
            completed = subprocess.run(command, capture_output=True, text=True, env=environment, timeout=30)
            readelf_parts.append("$ " + " ".join(command) + "\n" + completed.stdout + completed.stderr)
            require(completed.returncode == 0, f"readelf {option} failed: {completed.stderr.strip()}")
            if option == "-l":
                require(not re.search(r"^\s*INTERP\s", completed.stdout, re.MULTILINE),
                        f"The {role}executable requests a dynamic interpreter")
            if option == "-d":
                require("(NEEDED)" not in completed.stdout, f"The {role}executable requires a shared library")
        readelf_path = args.output.with_suffix(suffix)
        with readelf_path.open("x", encoding="utf-8") as stream:
            stream.write("\n".join(readelf_parts))
        record(report, role + "no_dynamic_interpreter_or_required_shared_libraries", evidence=str(readelf_path),
               readelf_sha256=sha256(readelf_path))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--archive", type=Path, required=True)
    parser.add_argument("--binary", type=Path, required=True)
    parser.add_argument("--worker", type=Path, help="Default: kindle-jpeg-worker beside --binary")
    parser.add_argument("--app-root", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True, help="A new JSON evidence file")
    parser.add_argument("--expected-binary-sha256", required=True)
    parser.add_argument("--expected-worker-sha256")
    parser.add_argument("--expected-archive-sha256")
    args = parser.parse_args()
    for name in ("expected_binary_sha256", "expected_archive_sha256", "expected_worker_sha256"):
        value = getattr(args, name)
        if value is not None and not re.fullmatch(r"[0-9a-fA-F]{64}", value):
            parser.error(f"--{name.replace('_', '-')} must be a SHA-256 hex digest")
    os.umask(0o077)
    args.output = args.output.resolve()
    args.output.parent.mkdir(parents=True, exist_ok=True)
    if any(path.exists() for path in (args.output, args.output.with_suffix(".readelf.txt"),
                                      args.output.with_suffix(".worker.readelf.txt"))):
        parser.error("The evidence paths already exist; select a new output file")
    report = {
        "schema_version": 1,
        "status": "failed",
        "passed": False,
        "started_at": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "coverage": "Static development-archive and ARM ELF checks only",
        "build_executed_by_checker": False,
        "arm_binary_executed": False,
        "device_verified": False,
        "checks": [],
    }
    try:
        check_package(args, report)
        report.update(status="passed", passed=True)
    except Exception as error:
        report["error"] = str(error)
    report["finished_at"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    with args.output.open("x", encoding="utf-8") as stream:
        json.dump(report, stream, indent=2)
        stream.write("\n")
    print(json.dumps({"status": report["status"], "output": str(args.output), "error": report.get("error")}))
    return 0 if report["passed"] else 1


if __name__ == "__main__":
    sys.exit(main())
